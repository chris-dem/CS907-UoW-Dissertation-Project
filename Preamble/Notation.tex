\newglossaryentry{not-replace}{
name={\ensuremath{x_{-i}(e) \equiv x(x_{-i}, e)}},
description={Given $x \in A^d$, we denote $x_{-i}(e)$ or $x(x_{-i}, e)$ to indicate $n$ dimensional vector, where
$$
    \forall j \in [n]: [x_{-i}(e)]_j =  [x(x_{-i}, e)]_j = \begin{cases}
        e   & \text{if } j = i \\
        x_i & \text{otherwise}
    \end{cases}
$$
Will be used to indicate element replacement},
type=symbols,
sort={0211}
}
